\begin{table}[h]
  \caption{Task 4 unknownness scores; larger $u(x)$ means more novel. $z_k$: logits, $p_k$: softmax probabilities, $K=10$; $f(x)$: 512-d penultimate feature, $\mu_c$: class means, $\Sigma$: shared diagonal covariance from CIFAR-10 training features with $\epsilon=10^{-6}$ added to each diagonal entry.}
  \label{tab:task4_formulas}
  \centering
  \small
  \begin{tabular}{lll}
    \toprule
    \textbf{Score} & \textbf{Unknownness} & \textbf{Captures} \\
    \midrule
    MSP         & $u(x) = 1-\max_k p_k(x)$ & normalised confidence \\
    MLS         & $u(x) = -\max_k z_k(x)$ & largest raw logit \\
    Energy      & $u(x) = -\log\sum_{k=1}^{K} e^{z_k(x)}$ & all logits \\
    Mahalanobis & $u(x) = \min_c\,(f(x)-\mu_c)^{\top}\Sigma^{-1}(f(x)-\mu_c)$ & distance to nearest class \\
    \bottomrule
  \end{tabular}
\end{table}
